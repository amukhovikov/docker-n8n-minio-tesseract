function hmac_sha256_hex(key, data) {

// 3. АЛГОРИТМ AWS SIGNATURE V4
const method = 'GET';
const endpoint = `/${bucket}/${filename}`;
const service = 's3';
const region = 'us-east-1';

const now = new Date();
const amzDate = now.toISOString().replace(/[:\-]|\.\d{3}/g, '');
const datestamp = amzDate.substring(0, 8);

const queryParams = [
  'X-Amz-Algorithm=AWS4-HMAC-SHA256',
  `X-Amz-Credential=${encodeURIComponent(`${accessKey}/${datestamp}/${region}/${service}/aws4_request`)}`,
  `X-Amz-Date=${amzDate}`,
  'X-Amz-Expires=900', // 15 минут
  'X-Amz-SignedHeaders=host'
].sort().join('&');

const canonicalRequest = `${method}\n${endpoint}\n${queryParams}\nhost:${host}\n\nhost\nUNSIGNED-PAYLOAD`;

const kDate = hmac_sha256(`AWS4${secretKey}`, datestamp);
const kRegion = hmac_sha256(kDate, region);
const kService = hmac_sha256(kRegion, service);
const kSigning = hmac_sha256(kService, 'aws4_request');

const stringToSign = `AWS4-HMAC-SHA256\n${amzDate}\n${datestamp}/${region}/${service}/aws4_request\n${sha256(canonicalRequest)}`;
const signature = hmac_sha256_hex(kSigning, stringToSign);

// 4. ИТОГОВЫЙ URL
const presignedUrl = `http://${host}${endpoint}?${queryParams}&X-Amz-Signature=${signature}`;

return [{ json: { url: presignedUrl } }];
